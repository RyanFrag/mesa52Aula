fotoFoiCancelada(erro: unknown): boolean {
  const codigo = (erro as { code?: string }).code;
  const texto = String((erro as { message?: string }).message ?? '').toLowerCase();

  return codigo === CameraErrorCode.TakePhotoCancelled

mensagemFoto(erro: unknown): string {
  const codigo = (erro as { code?: string }).code;

  if (codigo === CameraErrorCode.NoCameraAvailable) {
    return 'A câmera não está disponível neste aparelho.';
  }

  if (
    codigo === CameraErrorCode.CameraPermissionDenied
    || codigo === CameraErrorCode.GalleryPermissionDenied
  ) {
    return 'Permissão para usar a câmera ou a galeria foi negada.';
  }

  return 'Não foi possível obter a foto.';
}

Implementar tirarFoto

src/app/home/home.page.ts

Cole entre o logout() e o fotoFoiCancelada. Salve.

async tirarFoto(): Promise<void> {
  this.erroFoto = '';

  try {
    // Abre a câmera nativa do dispositivo e aguarda uma foto.
    const resultado = await Camera.takePhoto({
      quality: 80,
      saveToGallery: false,
    });

    // O webPath pode ser usado diretamente como src da imagem.
    if (resultado.webPath) {
      this.fotoMesa = resultado.webPath;
    }
  } catch (erro) {
    if (this.fotoFoiCancelada(erro)) {
      return;
    }

    console.error('Erro ao tirar foto:', erro);
    this.erroFoto = this.mensagemFoto(erro);
  }
}

Leia nesta ordem, em voz alta

  1. this.erroFoto = '' apaga a mensagem da tentativa anterior.
  2. Camera.takePhoto abre a câmera e espera. O await segura a função até a foto voltar ou a pessoa cancelar.
  3. quality: 80 é a compressão do JPEG, de 0 a 100.
  4. saveToGallery: false não grava cópia na galeria. Por isso a aula não pede permissão de armazenamento.
  5. webPath é o endereço que a tag img entende. Guarde na propriedade. Ainda não há img na tela: isso é o passo 17.
  6. Se a pessoa cancelar, fotoFoiCancelada dá return. A foto antiga permanece.
  7. Erro de verdade vai para o console.error. A tela recebe só a frase de mensagemFoto.

Implementar escolherDaGaleria

O mesmo plugin. Não instale outro pacote. takePhoto abre a câmera e devolve um objeto. chooseFromGallery abre a galeria e devolve uma lista.

src/app/home/home.page.ts

Onde achar

Logo abaixo da chave que fecha tirarFoto(). Cole e salve.

async escolherDaGaleria(): Promise<void> {
  this.erroFoto = '';

  try {
    // chooseFromGallery abre a galeria. Aqui só uma imagem é permitida.
    const escolha = await Camera.chooseFromGallery({
      quality: 80,
      allowMultipleSelection: false,
    });

    const foto = escolha.results[0];

    // O webPath pode ser usado diretamente como src da imagem.
    if (foto?.webPath) {
      this.fotoMesa = foto.webPath;
    }
  } catch (erro) {
    if (this.fotoFoiCancelada(erro)) {
      return;
    }

    console.error('Erro ao escolher foto:', erro);
    this.erroFoto = this.mensagemFoto(erro);
  }
}

allowMultipleSelection: false pede uma imagem. results[0] é essa imagem. O webPath dela vai para a mesma propriedade fotoMesa, então o preview é o mesmo img do passo 17.

Colar a frase de erro do GPS

src/app/home/home.page.ts

Cole no final da classe, antes da última chave que fecha HomePage. Salve.

mensagemLocalizacao(erro: unknown): string {
  const codigo = String((erro as { code?: number | string }).code ?? '');
  const texto = String((erro as { message?: string }).message ?? '').toLowerCase();

  if (codigo.includes('0018') || texto.includes('not declared')) {
    return 'Falta a permissão de localização no AndroidManifest.';
  }

  if (codigo === '1' || codigo.includes('0003') || texto.includes('denied') || texto.includes('permiss')) {
    return 'Permissão de localização negada.';
  }

  if (codigo === '3' || codigo.includes('0010') || texto.includes('timeout') || texto.includes('in time')) {
    return 'A localização demorou demais. Ligue o GPS e tente de novo.';
  }

  if (
    codigo === '2'
    || codigo.includes('0007')
    || texto.includes('disabled')
    || texto.includes('not enabled')
    || texto.includes('turned off')
    || texto.includes('unavailable')
  ) {
    return 'A localização do aparelho parece desativada.';
  }

  return 'Não foi possível obter sua localização.';
}

No Chrome o código é 1, 2 ou 3. No Android o código vem como OS-PLUG-GLOC-0010. Por isso a função também olha o texto da mensagem. A stack fica no console, no próximo passo.

Implementar obterLocalizacao

src/app/home/home.page.ts

Cole abaixo de escolherDaGaleria() e salve.

async obterLocalizacao(): Promise<void> {
  this.erroLocalizacao = '';
  this.carregandoLocalizacao = true;

  try {
    // No celular, estas chamadas consultam e pedem a permissão.
    // No navegador, requestPermissions não existe: o aviso aparece no getCurrentPosition.
    try {
      const permissao = await Geolocation.checkPermissions();
      if (permissao.location !== 'granted') {
        const pedido = await Geolocation.requestPermissions();
        if (pedido.location !== 'granted') {
          this.erroLocalizacao = 'Permissão de localização negada.';
          return;
        }
      }
    } catch (erroPermissao) {
      console.error('Permissão de localização:', erroPermissao);
    }

    // O GPS fino demora em sala fechada.
    // Se estourar o tempo, lerPosicao pede a localização da rede.
    const posicao = await this.lerPosicao();

    // As coordenadas ficam dentro de coords.
    this.latitude = posicao.coords.latitude;
    this.longitude = posicao.coords.longitude;
    // A precisão é informada em metros.
    this.precisao = posicao.coords.accuracy;
  } catch (erro) {
    console.error('Erro ao obter localização:', erro);
    this.erroLocalizacao = this.mensagemLocalizacao(erro);
  } finally {
    this.carregandoLocalizacao = false;
  }
}

async lerPosicao() {
  try {
    // Solicita ao dispositivo a posição GPS atual.
    return await Geolocation.getCurrentPosition({
      enableHighAccuracy: true,
      timeout: 20000,
    });
  } catch (erroGps) {
    const texto = String((erroGps as { message?: string }).message ?? '').toLowerCase();
    const codigo = String((erroGps as { code?: string | number }).code ?? '');
    const estourouTempo = codigo.includes('0010')
      || texto.includes('timeout')
      || texto.includes('in time');

    if (!estourouTempo) {
      throw erroGps;
    }

    return await Geolocation.getCurrentPosition({
      enableHighAccuracy: false,
      timeout: 20000,
    });
  }
}

Leia nesta ordem

  1. Limpa o erro e liga o carregamento.
  2. No Android, checkPermissions e requestPermissions consultam e abrem o diálogo. Se a pessoa negar, a função para aí.
  3. No Chrome, requestPermissions não existe e cai no catch de dentro. O aviso do site aparece na leitura da posição.
  4. lerPosicao pede o GPS fino e espera 20 segundos. Em sala fechada isso estoura. A segunda chamada usa a rede, com enableHighAccuracy: false.
  5. Os números estão em posicao.coords. accuracy é metros.
  6. finally desliga o spinner no sucesso e no erro. Sem ele, o botão fica desabilitado para sempre.
🚨

<uses-permission android:name="android.permission.ACCESS_COARSE_LOCATION" />
<uses-permission android:name="android.permission.ACCESS_FINE_LOCATION" />
  • ACCESS_COARSE_LOCATION é a localização aproximada.
  • ACCESS_FINE_LOCATION é a localização mais precisa. Combina com enableHighAccuracy: true.

Não cole READ_EXTERNAL_STORAGE, WRITE_EXTERNAL_STORAGE nem READ_MEDIA_IMAGES. saveToGallery está false. A câmera abre pelo aplicativo do sistema.
